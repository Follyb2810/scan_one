const nouns = ['Fox', 'Wolf', 'Hawk', 'Bear', 'Lion', 'Deer', 'Viper', 'Eagle', 'Tiger', 'Shark'];

const generateUsername = async () => {
  let username;
  const maxAttempts = 10;
  let attempts = 0;

  do {
    const adjective = adjectives[Math.floor(Math.random() * adjectives.length)];
    const noun = nouns[Math.floor(Math.random() * nouns.length)];
    const number = Math.floor(Math.random() * 1000);
    username = `${adjective}${noun}${number}`;
    attempts++;
  } while (await User.findOne({ username }) && attempts < maxAttempts);

  if (attempts >= maxAttempts) throw new Error('Unable to generate unique username');
  return username;
};

// Verify Signatures
const verifyEthSignature = async (walletAddress, signature, message) => {
  const signer = ethers.utils.verifyMessage(message, signature);
  return signer.toLowerCase() === walletAddress.toLowerCase();
};

const verifySolanaSignature = async (walletAddress, signature, message) => {
  const connection = new Connection('https://api.devnet.solana.com');
  const publicKey = new PublicKey(walletAddress);
  const encodedMessage = new TextEncoder().encode(message);
  return nacl.sign.detached.verify(encodedMessage, Buffer.from(signature, 'hex'), publicKey.toBuffer());

const checkQuota = async (username, action) => {
  const user = await User.findOne({ username });
  if (user.subscription.status && user.subscription.expiry > new Date()) {
    return true; // Subscribed users have no limits

  const today = new Date().setHours(0, 0, 0, 0);

  if (!usage) {
    usage = new Usage({ username, date: today, emailCount: 0, eventCount: 0 });

    usage.eventCount++;
  } else {
    action === 'email' ? usage.emailCount++ : usage.eventCount++;
  }

  await usage.save();
  return true;

### Implementation

#### Step 1: Smart Contract (Polygon)
A simple contract for subscriptions and micro-payments (USDC-based).

```solidity
pragma solidity ^0.8.0;

interface IERC20 {
    function transferFrom(address sender, address recipient, uint256 amount) external returns (bool);
}

contract Payments {
    address public owner;
    IERC20 public usdc; // USDC contract address
    uint256 public subscriptionFee = 1 * 10**6; // $1 in USDC (6 decimals)
    uint256 public perUseFee = 0.01 * 10**6; // $0.01 in USDC
    mapping(address => uint256) public subscriptions; // Expiry timestamp
    mapping(address => uint256) public balances; // Micro-payment balance

    event Subscribed(address user, uint256 expiry);
    event MicroPayment(address user, uint256 amount);

    constructor(address usdcAddress) {
        owner = msg.sender;
        usdc = IERC20(usdcAddress);
    }

    function subscribe() external {
        require(usdc.transferFrom(msg.sender, owner, subscriptionFee), "Payment failed");
        subscriptions[msg.sender] = block.timestamp + 30 days;
        emit Subscribed(msg.sender, subscriptions[msg.sender]);
    }

    function payPerUse() external {
        require(usdc.transferFrom(msg.sender, owner, perUseFee), "Payment failed");
        balances[msg.sender] += perUseFee;
        emit MicroPayment(msg.sender, perUseFee);
    }

    function withdraw() external {
        require(msg.sender == owner, "Not owner");
        uint256 balance = usdc.balanceOf(address(this));
        usdc.transferFrom(address(this), owner, balance);
    }

    function isSubscribed(address user) external view returns (bool) {
        return subscriptions[user] > block.timestamp;
    }

    function hasBalance(address user) external view returns (bool) {
        return balances[user] >= perUseFee;
    }
}
```

  const { username } = req.query;
  const emails = await Email.find({
    $or: [{ senderUsername: username }, { recipientUsername: username }],
  });
  const emailsWithBody = await Promise.all(emails.map(async email => {
    try {
      const response = await pinata.catJSON(email.bodyCid);
      return { ...email._doc, body: response.body };
    } catch {
      return { ...email._doc, body: 'Error fetching body' };
    }
  }));
  res.json(emailsWithBody);
});

// Calendar Routes
app.post('/api/events', async (req, res) => {
  const { username, title, start, end } = req.body;
  if (!(await checkQuota(username, 'event'))) {
    return res.status(403).json({ message: 'Event quota exceeded. Please subscribe or pay per use.' });
  }

  try {
    const eventData = { title, start, end };
    const pinataResponse = await pinata.pinJSONToIPFS(eventData, {
      pinataMetadata: { name: `event_${Date.now()}` },
    });
    const cid = pinataResponse.IpfsHash;

    const event = new Event({
      username,
      title,
      start,
      end,
      cid,
    });
    await event.save();
    res.status(201).json({ message: 'Event created' });
  } catch (error) {
    res.status(500).json({ message: 'Failed to upload to IPFS' });
  }
});

app.get('/api/events', async (req, res) => {
  const { username } = req.query;
  const events = await Event.find({ username });
  const eventsWithDetails = await Promise.all(events.map(async event => {
    try {
      const response = await pinata.catJSON(event.cid);
      return { ...event._doc, details: response };
    } catch {
      return { ...event._doc, details: {} };
    }
  }));
  res.json(eventsWithDetails);
});

app.listen(5000, () => console.log('Server running on port 5000'));
