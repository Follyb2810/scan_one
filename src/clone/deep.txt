### 3. Authentication with Social Login

**Passport Configuration (config/passport.js):**
```javascript
const passport = require('passport');
const GoogleStrategy = require('passport-google-oauth20').Strategy;
const FacebookStrategy = require('passport-facebook').Strategy;
const TwitterStrategy = require('passport-twitter').Strategy;
const User = require('../models/User');

passport.serializeUser((user, done) => {
  done(null, user.id);
});

passport.deserializeUser(async (id, done) => {
  try {
    const user = await User.findById(id);
    done(null, user);
  } catch (err) {
    done(err, null);
  }
});

// Google Strategy
passport.use(new GoogleStrategy({
  clientID: process.env.GOOGLE_CLIENT_ID,
  clientSecret: process.env.GOOGLE_CLIENT_SECRET,
  callbackURL: '/auth/google/callback'
}, async (accessToken, refreshToken, profile, done) => {
  try {
    let user = await User.findOne({ 'socialAuth.googleId': profile.id });
    
    if (!user) {
      user = new User({
        username: profile.displayName,
        email: profile.emails[0].value,
        'socialAuth.googleId': profile.id,
        profilePicture: profile.photos[0].value
      });
      await user.save();
    }
    
    done(null, user);
  } catch (err) {
    done(err, null);
  }
}));

// Similar strategies for Facebook and Twitter
```

```javascript
const Stream = require('../models/Stream');

const cloudinary = require('cloudinary').v2;

// Configure Cloudinary
cloudinary.config({
  cloud_name: process.env.CLOUDINARY_CLOUD_NAME,
  api_key: process.env.CLOUDINARY_API_KEY,
  api_secret: process.env.CLOUDINARY_API_SECRET

exports.createStream = async (req, res) => {
  try {
    const { title, description } = req.body;
    const videoFile = req.files.video;
    
    // Upload video to Cloudinary with stream-only option
    const result = await cloudinary.uploader.upload(videoFile.tempFilePath, {
      resource_type: "video",
      eager: [
        { width: 300, height: 300, crop: "pad", audio_codec: "none" }, 
        { width: 160, height: 100, crop: "crop", gravity: "south", audio_codec: "none" }
      ],
      eager_async: true,
      type: "private", // Makes video stream-only
      invalidate: true
    });
    
    const stream = new Stream({
      user: req.user.id,
      title,
      description,
      videoUrl: result.secure_url,
      thumbnailUrl: result.eager[0].secure_url
    });
    
    await stream.save();
    res.status(201).json(stream);
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: 'Server error' });
  }
};
```

### 5. Real-time Chat with Socket.io

**Socket Setup (server.js):**
```javascript
const express = require('express');
const http = require('http');
const socketIo = require('socket.io');
const app = express();
const server = http.createServer(app);
const io = socketIo(server);

// Socket.io connection
io.on('connection', (socket) => {
  console.log('New client connected');
  
  // Join room for private messaging
  socket.on('join room', (room) => {
    socket.join(room);
  });
  
  // Handle chat messages
  socket.on('chat message', async ({ room, message, sender }) => {
    // Save message to DB
    const newMessage = new Message({
      room,
      sender,
      text: message,
      createdAt: new Date()
    });
    await newMessage.save();
    
    // Emit to room
    io.to(room).emit('new message', newMessage);
    
    // Send notification
    io.emit('notification', {
      recipient: room.replace(sender, '').replace('_', ''),
      message: `New message from ${sender}`
    });
  });
  
  socket.on('disconnect', () => {
    console.log('Client disconnected');
  });
});

server.listen(PORT, () => console.log(`Server running on port ${PORT}`));
```

### 4. Real-time Chat Implementation

**Chat Screen (screens/Chat/ChatScreen.js):**
```javascript
import React, { useState, useEffect, useRef } from 'react';
import { View, Text, TextInput, FlatList, StyleSheet, KeyboardAvoidingView } from 'react-native';
import io from 'socket.io-client';
import axios from 'axios';

const ChatScreen = ({ route }) => {
  const { roomId, recipient } = route.params;
  const [messages, setMessages] = useState([]);
  const [newMessage, setNewMessage] = useState('');
  const socket = useRef(null);
  const flatListRef = useRef(null);
  
  useEffect(() => {
    // Connect to Socket.io
    socket.current = io('http://your-backend-url');
    
    // Join the room
    socket.current.emit('join room', roomId);
    
    // Load previous messages
    const loadMessages = async () => {
      try {
        const response = await axios.get(`/api/chat/${roomId}`);
        setMessages(response.data);
      } catch (err) {
        console.error(err);
      }
    };
    
    loadMessages();
    
    // Listen for new messages
    socket.current.on('new message', (message) => {
      setMessages(prev => [...prev, message]);
    });
    
    return () => {
      socket.current.disconnect();
    };
  }, [roomId]);
  
  const sendMessage = () => {
    if (newMessage.trim()) {
      const messageData = {
        room: roomId,
        message: newMessage,
        sender: 'currentUserId' // Replace with actual user ID
      };
      
      socket.current.emit('chat message', messageData);
      setNewMessage('');
    }
  };
  
  return (
    <KeyboardAvoidingView 
      style={styles.container}
      behavior="padding"
      keyboardVerticalOffset={90}
    >
      <FlatList
        ref={flatListRef}
        data={messages}
        keyExtractor={(item) => item._id}
        renderItem={({ item }) => (
          <View style={[
            styles.messageBubble,
            item.sender === 'currentUserId' ? styles.sentMessage : styles.receivedMessage
          ]}>
            <Text style={styles.messageText}>{item.text}</Text>
          </View>
        )}
        onContentSizeChange={() => flatListRef.current.scrollToEnd({ animated: true })}
        style={styles.messagesList}
      />
      
      <View style={styles.inputContainer}>
        <TextInput
          style={styles.input}
          value={newMessage}
          onChangeText={setNewMessage}
          placeholder="Type a message..."
          onSubmitEditing={sendMessage}
        />
        <Button title="Send" onPress={sendMessage} />
      </View>
    </KeyboardAvoidingView>
  );
};

const styles = StyleSheet.create({

    flex: 1,
  },
```

### 5. Virtual Gifting System

**Gift Shop Component (components/stream/GiftShop.js):**
```javascript
import React, { useState } from 'react';
import { View, Text, FlatList, TouchableOpacity, StyleSheet, Image } from 'react-native';
import axios from 'axios';

const gifts = [
  { id: 1, name: 'Rose', price: 10, image: require('../assets/gifts/rose.png') },
  { id: 2, name: 'Diamond', price: 50, image: require('../assets/gifts/diamond.png') },
  { id: 3, name: 'Car', price: 100, image: require('../assets/gifts/car.png') },
  { id: 4, name: 'Plane', price: 500, image: require('../assets/gifts/plane.png') },
];

const GiftShop = ({ streamId, userId }) => {
  const [userCoins, setUserCoins] = useState(0);
  
  useEffect(() => {
    const fetchUserCoins = async () => {
      try {
        const response = await axios.get(`/api/users/${userId}/coins`);
        setUserCoins(response.data.coins);
      } catch (err) {
        console.error(err);
      }
    };
    
    fetchUserCoins();
  }, [userId]);
  
  const sendGift = async (gift) => {
    if (userCoins < gift.price) {
      alert('Not enough coins!');
      return;
    }
    
    try {
      await axios.post('/api/gifts/send', {
        streamId,
        senderId: userId,
        giftId: gift.id,
        amount: gift.price
      });
      
      // Update local coins
      setUserCoins(prev => prev - gift.price);
      
      // Show animation (would be implemented with Lottie or similar)
      alert(`You sent a ${gift.name}!`);
    } catch (err) {
      console.error(err);
      alert('Failed to send gift');
    }
  };
  
  return (
    <View style={styles.container}>
      <Text style={styles.title}>Send a Gift</Text>
      <Text style={styles.coins}>Your coins: {userCoins}</Text>
      
      <FlatList
        data={gifts}
        numColumns={2}
        keyExtractor={item => item.id.toString()}
        renderItem={({ item }) => (
          <TouchableOpacity 
            style={styles.giftItem}
            onPress={() => sendGift(item)}
            disabled={userCoins < item.price}
          >
            <Image source={item.image} style={styles.giftImage} />
            <Text style={styles.giftName}>{item.name}</Text>
            <Text style={styles.giftPrice}>{item.price} coins</Text>
          </TouchableOpacity>
        )}
      />
    </View>
  );
};
```

## Deployment Considerations

1. **Backend Deployment**:
   - Use a platform like Heroku, AWS, or DigitalOcean
   - Set up environment variables for sensitive data
   - Implement proper CORS configuration

2. **Database Deployment**:
   - MongoDB Atlas for cloud-based MongoDB
   - Set up proper indexing for performance
   - Implement regular backups

3. **Frontend Deployment**:
   - For iOS: Deploy to App Store
   - For Android: Deploy to Google Play Store
   - Use CodePush for over-the-air updates

4. **Video Streaming Optimization**:
   - Consider using a CDN for video delivery
   - Implement adaptive bitrate streaming
   - Set up proper caching headers

## Security Considerations

1. Implement JWT authentication with refresh tokens
2. Use HTTPS for all API calls
3. Sanitize user inputs to prevent XSS and injection attacks
4. Implement rate limiting on API endpoints
5. Use proper permissions for video access (stream-only)
6. Regularly update dependencies to patch vulnerabilities

This implementation provides a complete foundation for your TikTok clone with all the requested features. You'll need to customize it further based on your specific requirements and design preferences.
