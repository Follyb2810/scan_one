import React, { useState, useEffect } from 'react';
import { useAccount, useSignMessage, useContractWrite } from 'wagmi';
import axios from 'axios';
import FullCalendar from '@fullcalendar/react';
import dayGridPlugin from '@fullcalendar/daygrid';
import { Container, TextField, Button, List, ListItem, Typography, Dialog, DialogTitle, DialogContent } from '@mui/material';

const App = () => {
  const { address, isConnected, chain } = useAccount();
  const { signMessageAsync } = useSignMessage();
  const [username, setUsername] = useState('');
  const [emails, setEmails] = useState([]);
  const [events, setEvents] = useState([]);
  const [recipient, setRecipient] = useState('');
  const [subject, setSubject] = useState('');
  const [body, setBody] = useState('');
  const [newWallet, setNewWallet] = useState('');
  const [openDialog, setOpenDialog] = useState(false);

  const { write: subscribe } = useContractWrite({
    address: 'YOUR_CONTRACT_ADDRESS',
    abi: [
      {
        name: 'subscribe',
        type: 'function',
        inputs: [],
        outputs: [],
        stateMutability: 'payable',
      },
    ],
    functionName: 'subscribe',
    value: ethers.utils.parseEther('0.001'), // Adjust for USDC
  });

  useEffect(() => {
    if (isConnected && address) {
      registerUser();
      fetchEmails();
      fetchEvents();
    }
  }, [isConnected, address]);

  const registerUser = async () => {
    const message = 'Sign to register with Web3 Email';
    const signature = await signMessageAsync({ message });
    const res = await axios.post('http://localhost:5000/api/users/register', {
      walletAddress: address,
      signature,
      chain: chain?.name.toLowerCase(),
    });
    setUsername(res.data.username);
  };

  const addWallet = async () => {
    const message = 'Sign to add wallet';
    const signature = await signMessageAsync({ message });
    await axios.post('http://localhost:5000/api/users/add-wallet', {
      username,
      walletAddress: newWallet,
      signature,
      chain: chain?.name.toLowerCase(),
    });
    setNewWallet('');
    setOpenDialog(false);
  };

  const updateUsername = async () => {
    const newUsername = prompt('Enter new username (3-20 characters)');
    if (newUsername) {
      await axios.post('http://localhost:5000/api/users/username', {
        username,
        newUsername,
      });
      setUsername(newUsername);
    }
  };

  const handleSubscribe = async () => {
    await subscribe();
    await axios.post('http://localhost:5000/api/payments/subscribe', {
      username,
      walletAddress: address,
    });
  };

  const fetchEmails = async () => {
    const res = await axios.get('http://localhost:5000/api/emails', {
      params: { username },
    });
    setEmails(res.data);
  };

  const fetchEvents = async () => {
    const res = await axios.get('http://localhost:5000/api/events', {
      params: { username },
    });
    setEvents(res.data.map(event => ({
      id: event._id,
      title: event.title,
      start: event.start,
      end: event.end,
    })));
  };

  const sendEmail = async () => {
    try {
      await axios.post('http://localhost:5000/api/emails', {
        senderUsername: username,
        recipientUsername: recipient,
        subject,
        body,
      });
      fetchEmails();
      setRecipient('');
      setSubject('');
      setBody('');
    } catch (error) {
      alert(error.response?.data?.message || 'Failed to send email');
    }
  };

  return (
    <Container>
      {!isConnected ? (
        <Typography>Connect your wallet to continue</Typography>
      ) : (
        <>
          <Typography variant="h6">Wallet: {address}</Typography>
          <Typography variant="h6">Username: {username}</Typography>
          <Button onClick={updateUsername} variant="outlined">Update Username</Button>
          <Button onClick={() => setOpenDialog(true)} variant="outlined">Add Wallet</Button>
          <Button onClick={handleSubscribe} variant="contained">Subscribe ($1/month)</Button>
          <Typography variant="h4">Email Client</Typography>
          <TextField
            label="Recipient Username"
            value={recipient}
            onChange={e => setRecipient(e.target.value)}
            fullWidth
          />
          <TextField
            label="Subject"
            value={subject}
            onChange={e => setSubject(e.target.value)}
            fullWidth
          />
          <TextField
            label="Body"
            value={body}
            onChange={e => setBody(e.target.value)}
            multiline
            rows={4}
            fullWidth
          />
          <Button onClick={sendEmail} variant="contained">Send Email</Button>
          <List>
            {emails.map(email => (
              <ListItem key={email._id}>
                {email.subject} - From: {email.senderUsername}
              </ListItem>
            ))}
          </List>
          <Typography variant="h4">Calendar</Typography>
          <FullCalendar
            plugins={[dayGridPlugin]}
            initialView="dayGridMonth"
            events={events}
          />
          <Dialog open={openDialog} onClose={() => setOpenDialog(false)}>
            <DialogTitle>Add New Wallet</DialogTitle>
            <DialogContent>
              <TextField
                label="Wallet Address"
                value={newWallet}
                onChange={e => setNewWallet(e.target.value)}
                fullWidth
              />
              <Button onClick={addWallet} variant="contained">Add</Button>
            </DialogContent>
          </Dialog>
        </>
      )}
    </Container>
  );
};

export default App;
</xaiArtefact>
```

<xaiArtefact artefact_id="b610f901-1225-44fb-9144-dece201581be" title="App.js" contentType="text/javascript">
import React, { useState, useEffect } from 'react';
import { View, Text, TextInput, Button, FlatList } from 'react-native';
import axios from 'axios';
import { Calendar } from 'react-native-calendars';

const App = () => {
  const [walletAddress, setWalletAddress] = useState('');
  const [username, setUsername] = useState('');
  const [emails, setEmails] = useState([]);
  const [events, setEvents] = useState([]);
  const [recipient, setRecipient] = useState('');
  const [subject, setSubject] = useState('');
  const [body, setBody] = useState('');

  useEffect(() => {
    if (walletAddress) {
      fetchEmails();
      fetchEvents();
    }
  }, [walletAddress]);

  const connectWallet = async () => {
    // Implement WalletConnect
    setWalletAddress('mock_wallet_address');
    const res = await axios.post('http://localhost:5000/api/users/register', {
      walletAddress: 'mock_wallet_address',
      signature: 'mock_signature',
      chain: 'ethereum',
    });
    setUsername(res.data.username);
  };

  const fetchEmails = async () => {
    const res = await axios.get('http://localhost:5000/api/emails', {
      params: { username },
    });
    setEmails(res.data);
  };

  const fetchEvents = async () => {
    const res = await axios.get('http://localhost:5000/api/events', {
      params: { username },
    });
    setEvents(res.data);
  };

  const sendEmail = async () => {
    try {
      await axios.post('http://localhost:5000/api/emails', {
        senderUsername: username,
        recipientUsername: recipient,
        subject,
        body,
      });
      fetchEmails();
    } catch (error) {
      alert(error.response?.data?.message || 'Failed to send email');
    }
  };

  return (
    <View style={{ padding: 20 }}>
      {!walletAddress ? (
        <Button title="Connect Wallet" onPress={connectWallet} />
      ) : (
        <>
          <Text>Wallet: {walletAddress}</Text>
          <Text>Username: {username}</Text>
          <Text>Email Client</Text>
          <TextInput
            placeholder="Recipient Username"
            value={recipient}
            onChangeText={setRecipient}
          />
          <TextInput
            placeholder="Subject"
            value={subject}
            onChangeText={setSubject}
          />
          <TextInput
            placeholder="Body"
            value={body}
            onChangeText={setBody}
            multiline
          />
          <Button title="Send Email" onPress={sendEmail} />
          <FlatList
            data={emails}
            keyExtractor={item => item._id}
            renderItem={({ item }) => (
              <Text>{item.subject} - From: {item.senderUsername}</Text>
            )}
          />
          <Text>Calendar</Text>
          <Calendar
            markedDates={events.reduce((acc, event) => ({
              ...acc,
              [event.start.split('T')[0]]: { marked: true },
            }), {})}
          />
        </>
      )}
    </View>
  );
};

export default App;
