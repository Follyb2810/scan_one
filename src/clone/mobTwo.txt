import React, { useState, useEffect } from 'react';
import { View, Text, TextInput, Button, FlatList } from 'react-native';
import axios from 'axios';
import { Calendar } from 'react-native-calendars';
// Note: WalletConnect integration requires additional setup (e.g., react-native-walletconnect)

const App = () => {
  const [walletAddress, setWalletAddress] = useState('');
  const [username, setUsername] = useState('');
  const [emails, setEmails] = useState([]);
  const [events, setEvents] = useState([]);
  const [recipient, setRecipient] = useState('');
  const [subject, setSubject] = useState('');
  const [body, setBody] = useState('');

  useEffect(() => {
    if (walletAddress) {
      fetchEmails();
      fetchEvents();
    }
  }, [walletAddress]);

  const connectWallet = async () => {
    // Implement WalletConnect logic here
    setWalletAddress('mock_wallet_address'); // Replace with actual wallet address
    const res = await axios.post('http://localhost:5000/api/users/register', {
      walletAddress: 'mock_wallet_address',
      signature: 'mock_signature',
      chain: 'ethereum',
    });
    setUsername(res.data.username);
  };

  const fetchEmails = async () => {
    const res = await axios.get('http://localhost:5000/api/emails', {
      params: { walletAddress },
    });
    setEmails(res.data);
  };

  const fetchEvents = async () => {
    const res = await axios.get('http://localhost:5000/api/events', {
      params: { walletAddress },
    });
    setEvents(res.data);
  };

  const sendEmail = async () => {
    await axios.post('http://localhost:5000/api/emails', {
      senderWallet: walletAddress,
      recipientUsername: recipient,
      subject,
      body,
    });
    fetchEmails();
  };

  return (
    <View style={{ padding: 20 }}>
      {!walletAddress ? (
        <Button title="Connect Wallet" onPress={connectWallet} />
      ) : (
        <>
          <Text>Wallet: {walletAddress}</Text>
          <Text>Username: {username}</Text>
          <Text>Email Client</Text>
          <TextInput
            placeholder="Recipient Username"
            value={recipient}
            onChangeText={setRecipient}
          />
          <TextInput
            placeholder="Subject"
            value={subject}
            onChangeText={setSubject}
          />
          <TextInput
            placeholder="Body"
            value={body}
            onChangeText={setBody}
            multiline
          />
          <Button title="Send Email" onPress={sendEmail} />
          <FlatList
            data={emails}
            keyExtractor={item => item._id}
            renderItem={({ item }) => (
              <Text>{item.subject} - From: {item.senderWallet}</Text>
            )}
          />
          <Text>Calendar</Text>
          <Calendar
            markedDates={events.reduce((acc, event) => ({
              ...acc,
              [event.start.split('T')[0]]: { marked: true },
            }), {})}
          />
        </>
      )}
    </View>
  );
};

export default App;
