```javascript
// config/passport.config.js

const passport = require('passport');
const LocalStrategy = require('passport-local').Strategy;
const GoogleStrategy = require('passport-google-oauth20').Strategy;
const FacebookStrategy = require('passport-facebook').Strategy;
const User = require('../models/user.model');
const bcrypt = require('bcryptjs');

// Serialize and Deserialize User
passport.serializeUser((user, done) => {
  done(null, user.id);
});

passport.deserializeUser(async (id, done) => {
  try {
    const user = await User.findById(id);
    done(null, user);
  } catch (err) {
    done(err, null);
  }
});

// Local Strategy
passport.use(
  new LocalStrategy(
    {
      usernameField: 'email',
      passwordField: 'password'
    },
    async (email, password, done) => {
      try {
        // Find user by email
        const user = await User.findOne({ email });
        
        // Check if user exists
        if (!user) {
          return done(null, false, { message: 'User not found' });
        }
        
        // Validate password
        const isMatch = await bcrypt.compare(password, user.password);
        if (!isMatch) {
          return done(null, false, { message: 'Incorrect password' });
        }
        
        // Return user if authentication is successful
        return done(null, user);
      } catch (err) {
        return done(err);
      }
    }
  )
);

// Google OAuth Strategy
passport.use(
  new GoogleStrategy(
    {
      clientID: process.env.GOOGLE_CLIENT_ID,
      clientSecret: process.env.GOOGLE_CLIENT_SECRET,
      callbackURL: '/api/auth/google/callback',
      scope: ['profile', 'email']
    },
    async (accessToken, refreshToken, profile, done) => {
      try {
        // Check if user already exists
        const existingUser = await User.findOne({ 'google.id': profile.id });
        
        if (existingUser) {
          return done(null, existingUser);
        }
        
        // Check if user exists with the same email
        const userWithEmail = await User.findOne({ email: profile.emails[0].value });
        
        if (userWithEmail) {
          // Link Google account to existing user
          userWithEmail.google = {
            id: profile.id,
            email: profile.emails[0].value,
            name: profile.displayName
          };
          await userWithEmail.save();
          return done(null, userWithEmail);
        }
        
        // Create a new user if it doesn't exist
        const newUser = new User({
          email: profile.emails[0].value,
          username: profile.displayName.replace(/\s+/g, '').toLowerCase() + Math.floor(Math.random() * 1000),
          fullName: profile.displayName,
          profilePicture: profile.photos[0].value,
          google: {
            id: profile.id,
            email: profile.emails[0].value,
            name: profile.displayName
          },
          isVerified: true
        });
        
        await newUser.save();
        return done(null, newUser);
      } catch (err) {
        return done(err, null);
      }
    }
  )
);

// Facebook OAuth Strategy
passport.use(
  new FacebookStrategy(
    {
      clientID: process.env.FACEBOOK_APP_ID,
      clientSecret: process.env.FACEBOOK_APP_SECRET,
      callbackURL: '/api/auth/facebook/callback',
      profileFields: ['id', 'emails', 'name', 'picture.type(large)']
    },
    async (accessToken, refreshToken, profile, done) => {
      try {
        // Check if user already exists
        const existingUser = await User.findOne({ 'facebook.id': profile.id });
        
        if (existingUser) {
          return done(null, existingUser);
        }
        
        // Check if email exists
        if (profile.emails && profile.emails.length > 0) {
          const userWithEmail = await User.findOne({ email: profile.emails[0].value });
          
          if (userWithEmail) {
            // Link Facebook account to existing user
            userWithEmail.facebook = {
              id: profile.id,
              email: profile.emails[0].value,
              name: `${profile.name.givenName} ${profile.name.familyName}`
            };
            await userWithEmail.save();
            return done(null, userWithEmail);
          }
        }
        
        // Create a new user
        const newUser = new User({
          email: profile.emails ? profile.emails[0].value : `${profile.id}@facebook.com`,
          username: `fb_${profile.id.substring(0, 8)}${Math.floor(Math.random() * 1000)}`,
          fullName: `${profile.name.givenName} ${profile.name.familyName}`,
          profilePicture: profile.photos ? profile.photos[0].value : '',
          facebook: {
            id: profile.id,
            email: profile.emails ? profile.emails[0].value : null,
            name: `${profile.name.givenName} ${profile.name.familyName}`
          },
          isVerified: true
        });
        
        await newUser.save();
        return done(null, newUser);
      } catch (err) {
        return done(err, null);
      }
    }
  )
);

module.exports = passport;
```

```javascript
// routes/auth.routes.js

const express = require('express');
const router = express.Router();
const passport = require('passport');
const jwt = require('jsonwebtoken');
const bcrypt = require('bcryptjs');
const { v4: uuidv4 } = require('uuid');
const { User } = require('../models/user.model');
const { sendVerificationEmail, sendPasswordResetEmail } = require('../utils/email.util');
const { isAuthenticated } = require('../middleware/auth.middleware');

// Register a new user
router.post('/register', async (req, res) => {
  try {
    const { username, email, password, fullName } = req.body;

    // Check if user already exists
    const userExists = await User.findOne({ 
      $or: [{ email }, { username }] 
    });

    if (userExists) {
      return res.status(400).json({ 
        success: false, 
        message: userExists.email === email 
          ? 'Email already in use' 
          : 'Username already taken' 
      });
    }

    // Create verification token
    const verificationToken = uuidv4();

    // Create a new user
    const newUser = new User({
      username,
      email,
      password,
      fullName,
      resetPasswordToken: verificationToken,
      resetPasswordExpires: Date.now() + 3600000 // 1 hour
    });

    await newUser.save();

    // Send verification email
    await sendVerificationEmail(email, verificationToken);

    // Create JWT token
    const token = jwt.sign(
      { id: newUser._id },
      process.env.JWT_SECRET,
      { expiresIn: '30d' }
    );

    res.status(201).json({
      success: true,
      message: 'User registered successfully. Please check your email to verify your account.',
      token,
      user: {
        id: newUser._id,
        username: newUser.username,
        email: newUser.email,
        fullName: newUser.fullName,
        profilePicture: newUser.profilePicture,
        isVerified: newUser.isVerified
      }
    });
  } catch (err) {
    console.error('Registration error:', err);
    res.status(500).json({
      success: false,
      message: 'Registration failed',
      error: err.message
    });
  }
});

        message: 'Authentication error',

        username: user.username,
        email: user.email,
        fullName: user.fullName,
        profilePicture: user.profilePicture,
        isVerified: user.isVerified
```

```javascript
    const video = await Video.findById(videoId).populate('user');
    if (!video) {
      return res.status(404).json({
        success: false,
        message: 'Video not found'
      });
    }

    const sender = await User.findById(req.user._id);
    const recipient = video.user;

    // Check if sender has enough coins
    if (sender.coins < gift.coinValue) {
      return res.status(400).json({
        success: false,
        message: 'You don\'t have enough coins to send this gift'
      });
    }

    // Create gift transaction
    const giftTransaction = new GiftTransaction({
      sender: sender._id,
      recipient: recipient._id,
      gift: gift._id,
      video: video._id,
      coinAmount: gift.coinValue
    });

    await giftTransaction.save();

    // Update sender's coin balance
    sender.coins -= gift.coinValue;
    await sender.save();

    // Update recipient's coin balance (70% of the gift value)
    const recipientCoinAmount = Math.floor(gift.coinValue * 0.7);
    recipient.coins += recipientCoinAmount;
    await recipient.save();

    // Create coin transactions for sender and recipient
    const senderCoinTx = new CoinTransaction({
      user: sender._id,
      amount: -gift.coinValue,
      type: 'gift_sent',
      reference: `Gift to ${recipient.username} for video`
    });

    const recipientCoinTx = new CoinTransaction({
      user: recipient._id,
      amount: recipientCoinAmount,
      type: 'gift_received',
      reference: `Gift from ${sender.username} for video`
    });

    await senderCoinTx.save();
    await recipientCoinTx.save();

    // Create notification for recipient
    const notification = new Notification({
      recipient: recipient._id,
      sender: sender._id,
      type: 'gift',
      video: video._id,
      message: `${sender.username} sent you a gift: ${gift.name}`
    });

    await notification.save();

    res.status(200).json({
      success: true,

      giftTransaction

    res.status(500).json({

      message: 'Failed to send gift',
      error: err.message
```
