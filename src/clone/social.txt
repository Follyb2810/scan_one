For your TikTok clone app built with **Express.js** and **MongoDB**, implementing authentication with multiple options (Google, Facebook, QR code, and email/password) is a great way to provide flexibility to users. I’ll recommend libraries and provide examples for each authentication method. My preferred choice for ease of integration and scalability would be **Passport.js**, as it supports multiple strategies (Google, Facebook, etc.) and can be extended for custom flows like QR or email/password.

---

### Recommended Authentication Approach
- **Passport.js**: A widely-used middleware for Express.js that supports social logins (Google, Facebook) and can be customized for email/password or QR code flows.
- **Why Passport.js?**: It’s modular, supports multiple strategies out of the box, integrates well with Express.js, and works seamlessly with MongoDB for storing user data.

Install dependencies:
```bash
npm install express mongoose passport passport-google-oauth20 passport-facebook jsonwebtoken qrcode bcryptjs
```

Update `app.js`:
```javascript
const express = require('express');
const mongoose = require('mongoose');

require('dotenv').config();

const app = express();

```

### Step 3: Authentication Strategies
#### 1. Google Login (Using `passport-google-oauth20`)
**Setup**: Get Google Client ID and Secret from [Google Cloud Console](https://console.cloud.google.com/).
- Add to `.env`:
  ```
  GOOGLE_CLIENT_ID=your_google_client_id
  GOOGLE_CLIENT_SECRET=your_google_client_secret
  CALLBACK_URL=http://localhost:5000/api/auth/google/callback
  ```

**Passport Strategy** (`config/passport.js`):
```javascript
const passport = require('passport');
const GoogleStrategy = require('passport-google-oauth20').Strategy;
const User = require('../models/User');

passport.use(new GoogleStrategy({
  clientID: process.env.GOOGLE_CLIENT_ID,
  clientSecret: process.env.GOOGLE_CLIENT_SECRET,
  callbackURL: process.env.CALLBACK_URL
}, async (accessToken, refreshToken, profile, done) => {
  try {
    let user = await User.findOne({ googleId: profile.id });
    if (!user) {
      user = new User({
        googleId: profile.id,
        username: profile.displayName,
        email: profile.emails[0].value,
        profilePic: profile.photos[0].value
      });
      await user.save();
    }
    done(null, user);
  } catch (err) {
    done(err, null);
  }
}));

passport.serializeUser((user, done) => done(null, user.id));
passport.deserializeUser(async (id, done) => {
  const user = await User.findById(id);
  done(null, user);
});

module.exports = passport;
```

**Routes** (`routes/authRoutes.js`):
```javascript
const express = require('express');
const passport = require('passport');
const jwt = require('jsonwebtoken');
const router = express.Router();

router.get('/google', passport.authenticate('google', { scope: ['profile', 'email'] }));

router.get('/google/callback', passport.authenticate('google', { session: false }), (req, res) => {
  const token = jwt.sign({ id: req.user._id }, process.env.JWT_SECRET, { expiresIn: '1h' });
  res.redirect(`yourapp://auth?token=${token}`); // Redirect to app with token
});

module.exports = router;
```

#### 3. Email/Password Login (Custom with `bcryptjs`)
**Controller** (`controllers/authController.js`):
```javascript
const User = require('../models/User');
const bcrypt = require('bcryptjs');
const jwt = require('jsonwebtoken');

exports.register = async (req, res) => {
  const { username, email, password } = req.body;
  try {
    let user = await User.findOne({ email });
    if (user) return res.status(400).json({ message: 'User already exists' });

    const hashedPassword = await bcrypt.hash(password, 10);
    user = new User({ username, email, password: hashedPassword });
    await user.save();

    const token = jwt.sign({ id: user._id }, process.env.JWT_SECRET, { expiresIn: '1h' });
    res.status(201).json({ token });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
};

exports.login = async (req, res) => {
  const { email, password } = req.body;
  try {
    const user = await User.findOne({ email });
    if (!user || !await bcrypt.compare(password, user.password)) {
      return res.status(401).json({ message: 'Invalid credentials' });
    }
    const token = jwt.sign({ id: user._id }, process.env.JWT_SECRET, { expiresIn: '1h' });
    res.json({ token });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
};
```

**Routes** (`routes/authRoutes.js` - Add to existing file):
```javascript
const { register, login } = require('../controllers/authController');

router.post('/register', register);
router.post('/login', login);
```

#### 4. QR Code Login (Custom with `qrcode`)
**Idea**: Generate a QR code containing a unique token. The user scans it via the app, and the backend verifies it.
- Add to `.env`:
  ```
  JWT_SECRET=your_jwt_secret
  ```

**Controller** (`controllers/authController.js` - Add to existing file):
```javascript
const QRCode = require('qrcode');

exports.generateQR = async (req, res) => {
  try {
    const user = await User.findById(req.userId); // Assume user is authenticated
    if (!user) return res.status(404).json({ message: 'User not found' });

    const qrToken = jwt.sign({ id: user._id }, process.env.JWT_SECRET, { expiresIn: '5m' });
    user.qrCode = qrToken;
    await user.save();

    const qrCodeUrl = await QRCode.toDataURL(qrToken);
    res.json({ qrCodeUrl });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
};

exports.verifyQR = async (req, res) => {
  const { qrToken } = req.body;
  try {
    const decoded = jwt.verify(qrToken, process.env.JWT_SECRET);
    const user = await User.findById(decoded.id);
    if (!user || user.qrCode !== qrToken) {
      return res.status(401).json({ message: 'Invalid QR code' });
    }
    user.qrCode = null; // Invalidate after use
    await user.save();

    const token = jwt.sign({ id: user._id }, process.env.JWT_SECRET, { expiresIn: '1h' });
    res.json({ token });
  } catch (error) {
    res.status(401).json({ error: error.message });
  }
};
```

**Routes** (`routes/authRoutes.js` - Add to existing file):
```javascript
const { generateQR, verifyQR } = require('../controllers/authController');
const authMiddleware = require('../middleware/auth'); // JWT middleware (below)

router.get('/qr', authMiddleware, generateQR);
router.post('/qr/verify', verifyQR);
```

**JWT Middleware** (`middleware/auth.js`):
```javascript
const jwt = require('jsonwebtoken');

module.exports = (req, res, next) => {
  const token = req.header('Authorization')?.replace('Bearer ', '');
  if (!token) return res.status(401).json({ message: 'No token provided' });

  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    req.userId = decoded.id;
    next();
  } catch (error) {
    res.status(401).json({ message: 'Invalid token' });
  }
};
```

### Step 4: Frontend Integration (Android/iOS)
- **Google/Facebook**: Use OAuth libraries (e.g., `react-native-google-signin`, `react-native-fbsdk`) to initiate login and send the token to the backend.
- **Email/Password**: Simple form submission to `/register` and `/login` endpoints.
- **QR Code**: Display the QR code from `/qr` endpoint and scan it using a library like `react-native-qrcode-scanner`, then send the token to `/qr/verify`.

- **Google/Facebook**: Passport.js simplifies OAuth flows and handles token validation.
- **Email/Password**: Custom implementation with bcrypt for security and JWT for session management.
- **QR Code**: A creative, custom solution leveraging JWT and QR code generation for a unique login experience.
